cmake_minimum_required(VERSION 3.16)
project(forearm_chat_core LANGUAGES CXX)

add_library(forearm_chat_core INTERFACE)
target_include_directories(forearm_chat_core INTERFACE include)
target_compile_features(forearm_chat_core INTERFACE cxx_std_17)
